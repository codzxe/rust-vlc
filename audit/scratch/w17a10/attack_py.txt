import glob, json, os, re, sys

root, out = sys.argv[1], sys.argv[2]

# boundary -> (crate dirs that hold its parsers, untrusted?)
BOUNDARIES = [
    ("container demux",        ["crates/demux"],                      True),
    ("codec bitstream",        ["crates/codecs"],                     True),
    ("subtitles (ASS/SRT/VTT)",["crates/subtitles"],                  True),
    ("playlists (m3u/pls/cue/xspf)", ["crates/playlists"],            True),
    ("streaming (HLS/manifests/HTTP)", ["crates/streaming"],          True),
    ("source URI",             ["crates/input", "crates/media-types"], True),
    ("settings (TOML)",        ["crates/settings-schema", "crates/product-config"], True),
    ("control protocol",       ["crates/control-protocol", "apps/control-server"], True),
    ("wasm plugin ABI",        ["crates/plugin-host-wasm", "crates/plugin-api"], True),
    ("OS services (unsafe)",   ["crates/platform-windows", "crates/platform-linux", "crates/platform-macos"], False),
]

PREFIX = {
    "fuzz_demux": "container demux",
    "fuzz_codec": "codec bitstream",
    "fuzz_sub": "subtitles (ASS/SRT/VTT)",
    "fuzz_playlist": "playlists (m3u/pls/cue/xspf)",
    "fuzz_hls": "streaming (HLS/manifests/HTTP)",
    "fuzz_http": "streaming (HLS/manifests/HTTP)",
}

fuzz = {b: [] for b, _, _ in BOUNDARIES}
for p in glob.glob(os.path.join(root, "fuzz", "fuzz_targets", "*.rs")):
    name = os.path.basename(p)[:-3]
    for pre, b in PREFIX.items():
        if name.startswith(pre):
            fuzz[b].append(name)

# production hits already split by astgrep_summary.py (test code excluded)
hits = {b: {"panic": 0, "ignored": 0, "unwrap": 0, "noop_test": 0} for b, _, _ in BOUNDARIES}
j = os.path.join(root, "docs", "audit", "reports", "astgrep-summary.json")
if os.path.exists(j):
    for h in json.load(open(j, encoding="utf-8")).get("production_hits", []):
        f = h["file"]
        for b, dirs, _ in BOUNDARIES:
            if any(f.startswith(d + "/") for d in dirs):
                rule = h["rule"]
                if rule == "panic-in-lib":
                    hits[b]["panic"] += 1
                elif rule == "ignored-result":
                    hits[b]["ignored"] += 1
                elif rule == "unwrap-outside-tests":
                    hits[b]["unwrap"] += 1
                elif rule == "no-op-test":
                    hits[b]["noop_test"] += 1
                break

def loc(dirs):
    n = 0
    for d in dirs:
        for dp, _, fns in os.walk(os.path.join(root, d)):
            for fn in fns:
                if fn.endswith(".rs"):
                    try:
                        n += sum(1 for _ in open(os.path.join(dp, fn), encoding="utf-8", errors="replace"))
                    except OSError:
                        pass
    return n

lines = []
lines.append("# Attack surface matrix\n")
lines.append("Generated by `scripts/quality/22_attack_surface.sh` from evidence already")
lines.append("measured by T03 (ast-grep, test code excluded) and the repo's own fuzz")
lines.append("targets. **A boundary with 0 fuzz targets and shipped panics is a DoS.**\n")
lines.append("`panics` counts `panic!`/`assert!` in shipped code only. `unreachable!`")
lines.append("(1,827 workspace-wide, the bigger DoS surface) is ratcheted separately by")
lines.append("`24_panic_budget.sh`.\n")
lines.append("| boundary | untrusted | LOC | fuzz targets | panics | ignored `Result` | `unwrap` |")
lines.append("|---|---|---|---|---|---|---|")
order = sorted(BOUNDARIES, key=lambda b: (-len(fuzz[b[0]]), b[0]))
for b, dirs, untrusted in order:
    lines.append("| %s | %s | %s | %d | %d | %d | %d |" % (
        b, "**YES**" if untrusted else "no", "{:,}".format(loc(dirs)),
        len(fuzz[b]), hits[b]["panic"], hits[b]["ignored"], hits[b]["unwrap"]))

lines.append("\n## Gaps (a hacker starts here)\n")
for b, dirs, untrusted in order:
    if not untrusted:
        continue
    why = []
    if not fuzz[b]:
        why.append("**no fuzz target**")
    if hits[b]["panic"]:
        why.append("%d reachable panic site(s)" % hits[b]["panic"])
    if hits[b]["ignored"]:
        why.append("%d discarded `Result`(s)" % hits[b]["ignored"])
    if why:
        lines.append("- **%s**: %s" % (b, "; ".join(why)))
if not any(not fuzz[b] or hits[b]["panic"] or hits[b]["ignored"]
           for b, _, u in BOUNDARIES if u):
    lines.append("- none: every untrusted boundary has a fuzz target and no shipped panic.")

open(out, "w", encoding="utf-8").write("\n".join(lines) + "\n")
print("\n".join(lines))
